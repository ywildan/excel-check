' ============================================================
' Prototipe Audit Excel v0.1 — kode ThisWorkbook
' CARA PASANG:
'   1. Buka file .xlsm di Excel Windows, tekan Alt+F11
'   2. Di Project Explorer, klik dua kali "ThisWorkbook"
'   3. Hapus isi yang ada, lalu tempel SELURUH kode di bawah ini
'   4. Import juga modAudit.bas dan modIdentitas.bas via
'      File > Import File...
'   5. Tutup VBA Editor, Save (Ctrl+S), tutup lalu buka lagi file-nya
' ============================================================
Option Explicit

Private Sub Workbook_Open()
    ' Muat state lama (kalau file dibuka lagi setelah ditutup)
    LoadStateFromSheets

    If gIsFinished Then
        MsgBox "Tugas ini sudah SELESAI & TERKUNCI." & vbCrLf & _
               "Waktu aktif tercatat: " & FormatSeconds(gActiveSeconds), _
               vbInformation, "Audit Excel"
        UpdateDashboard
        Exit Sub
    End If

    If Not IsIdentityFilled() Then
        ThisWorkbook.Worksheets("Identitas").Activate
        MsgBox "Selamat datang di Prototipe Audit Excel v0.1" & vbCrLf & vbCrLf & _
               "1. Isi dulu sheet Identitas (Nama, NIM, Kelas, Matkul, Judul Tugas)." & vbCrLf & _
               "2. Tekan tombol MULAI di tab 'Tugas Akuntansi' di Ribbon." & vbCrLf & _
               "3. Kerjakan di sheet 'Jurnal Umum' seperti biasa." & vbCrLf & _
               "   Timer hanya jalan saat kamu aktif. Diam > 5 detik = jeda." & vbCrLf & _
               "4. Kalau selesai, tekan SELESAI & KUNCI.", _
               vbInformation, "Audit Excel"
    Else
        ' Identitas sudah ada (file dibuka lagi) — lanjutkan pelacakan otomatis
        StartTracking
    End If
    UpdateDashboard
End Sub

Private Sub Workbook_SheetChange(ByVal Sh As Object, ByVal Target As Range)
    ' Deteksi paste sederhana: CutCopyMode masih aktif saat change terjadi,
    ' atau sekali ubah > 1 sel. Bukan deteksi sempurna — di VSTO nanti
    ' akan dipakai event paste yang lebih akurat.
    Dim isPaste As Boolean
    isPaste = (Application.CutCopyMode <> False)
    RegisterChange Target, isPaste
    UpdateDashboard
End Sub

Private Sub Workbook_BeforeSave(ByVal SaveAsUI As Boolean, Cancel As Boolean)
    gSaveCount = gSaveCount + 1
    LogEvent "SIMPAN", "Save ke-" & gSaveCount & ". Aktif=" & FormatSeconds(gActiveSeconds)
    SaveStateToSheets
    SaveAuditToProperties
    UpdateDashboard
    ' Sembunyikan lagi sheet log kalau tadi dibuka via "Lihat Log"
    On Error Resume Next
    ThisWorkbook.Worksheets("_AUDIT_LOG").Visible = xlSheetVeryHidden
    On Error GoTo 0
End Sub

Private Sub Workbook_BeforeClose(Cancel As Boolean)
    If gIsTracking And Not gIsFinished Then
        LogEvent "TUTUP", "File ditutup. Aktif sejauh ini=" & FormatSeconds(gActiveSeconds) & _
                 ". Buka lagi untuk melanjutkan — angka tidak hilang."
    End If
    StopTracking
    SaveStateToSheets
    SaveAuditToProperties
    On Error Resume Next
    ThisWorkbook.Worksheets("_AUDIT_LOG").Visible = xlSheetVeryHidden
    On Error GoTo 0
End Sub
